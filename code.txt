    <script>
        function logEvent(message, type = 'info') {
            const logPanel = document.getElementById('log-panel');
            const time = new Date().toLocaleTimeString('en-US', { hour12: false, hour: "numeric", minute: "numeric", second: "numeric" });
            
            let color = '#94a3b8'; // default grey
            if (type === 'alert') color = '#f43f5e'; // red
            if (type === 'success') color = '#0ea5e9'; // blue
            if (type === 'warning') color = '#f59e0b'; // orange

            const logEntry = `<span style="color: ${color}">[${time}] ${message}</span><br>`;
            logPanel.innerHTML += logEntry;
            logPanel.scrollTop = logPanel.scrollHeight;
        }

        function setLineState(lineId, state) {
            const line = document.getElementById(lineId);
            if (state === 'attack') line.style.stroke = 'var(--attack-color)';
            else if (state === 'safe') line.style.stroke = 'var(--line-color)';
        }

        function simulateAttack() {
            logEvent("CRITICAL: Unauthorized payload detected from Public Client.", "alert");
            
            const external = document.getElementById('node-external');
            const gateway = document.getElementById('node-gateway');
            const db = document.getElementById('node-db');

            const mfaActive = document.getElementById('policy-mfa').checked;
            const microActive = document.getElementById('policy-micro').checked;

            // Attack begins
            external.className = 'node attacked';
            setLineState('line-1', 'attack');
            
            setTimeout(() => {
                if (!mfaActive) {
                    logEvent("ALERT: API Gateway compromised. Authentication bypassed.", "alert");
                    gateway.className = 'node attacked';
                    setLineState('line-2', 'attack');
                    
                    setTimeout(() => {
                        if (!microActive) {
                            logEvent("FATAL: Lateral movement successful. Database breached.", "alert");
                            db.className = 'node attacked';
                        } else {
                            logEvent("DEFENDED: Micro-segmentation isolated the threat. Database secure.", "warning");
                            db.className = 'node blocked';
                            setLineState('line-2', 'safe');
                        }
                    }, 1200);

                } else {
                    logEvent("DEFENDED: Request blocked by MFA policy. Payload dropped.", "warning");
                    gateway.className = 'node blocked';
                    setLineState('line-1', 'safe');
                }
            }, 1200);
        }

        function resetNetwork() {
            const nodes = document.querySelectorAll('.node');
            nodes.forEach(node => node.className = 'node safe');
            
            setLineState('line-1', 'safe');
            setLineState('line-2', 'safe');
            setLineState('line-3', 'safe');

            logEvent("System state restored. Resuming normal operations.", "success");
        }
